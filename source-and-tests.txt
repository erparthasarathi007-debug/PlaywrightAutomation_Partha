import { type Locator, type Page } from '@playwright/test';

export class ProductGridComponent {
  readonly cards: Locator;

  constructor(private readonly page: Page) {
    this.cards = page.locator('a[href*="/products/"]');
  }

  card(name: string): Locator {
    return this.cards.filter({
      has: this.page.getByRole('heading', { name, exact: true }),
    });
  }

  heading(name: string): Locator {
    return this.card(name).getByRole('heading', { name, exact: true });
  }

  async openProduct(name: string): Promise<void> {
    await this.card(name).click();
  }

  async names(): Promise<string[]> {
    return this.cards.getByRole('heading', { level: 3 }).allInnerTexts();
  }

  async prices(): Promise<string[]> {
    return this.cards.getByRole('heading', { level: 4 }).allInnerTexts();
  }
}

import { type Locator, type Page } from '@playwright/test';

export class CartPage {
  readonly quantityInput: Locator;
  readonly updateButton: Locator;
  readonly emptyMessage: Locator;

  constructor(private readonly page: Page) {
    this.quantityInput = page.locator('#cart input[id^="updates_"]');
    this.updateButton = page.getByRole('button', { name: 'Update', exact: true });
    this.emptyMessage = page.getByText(/cart is currently empty/i);
  }

  itemImage(name: string): Locator {
    return this.page.getByRole('img', { name: new RegExp(name, 'i') });
  }

  unitPrice(value: string): Locator {
    return this.page
      .locator('#cart .price:visible')
      .filter({ hasText: value });
  }

  total(value: string): Locator {
    return this.page.getByRole('heading', {
      name: `Total ${value}`,
      exact: true,
    });
  }

  async goto(): Promise<void> {
    await this.page.goto('/cart');
  }

  async setQuantity(value: string): Promise<void> {
    await this.quantityInput.fill(value);
  }

  async update(): Promise<void> {
    await this.updateButton.click();
  }
}

===== tests/e2e/storefront/add-to-cart.spec.ts =====
import { test, expect } from '../../../src/fixtures/test.fixture';

test.describe('Add to Cart', () => {
  test('updates quantity and removes the item at zero', async ({
    page, homePage, productPage, cartPage,
  }) => {
    await homePage.goto();
    await homePage.openProduct('Grey jacket');

    const productName = await productPage.heading('Grey jacket').innerText();
    const productPrice = await productPage.price('£55.00').innerText();
    expect(productName).toBe('Grey jacket');
    expect(productPrice).toBe('£55.00');

    await expect(homePage.header.cartSummary(0)).toBeVisible();
    await productPage.addToCart();
    await expect(homePage.header.cartSummary(1)).toBeVisible();

    // The header summary changes the hash; it does not navigate to the cart.
    await homePage.header.cartSummary(1).click();
    await expect(page).toHaveURL(/#$/);
    await cartPage.goto();

    await expect(page).toHaveURL(/\/cart/);
    await expect(homePage.header.cartSummary(1)).toBeVisible();
    await expect(cartPage.itemImage(productName)).toBeVisible();
    await expect(cartPage.unitPrice(productPrice).first()).toBeVisible();
    await expect(cartPage.quantityInput).toHaveValue('1');

    await cartPage.setQuantity('2');
    await cartPage.update();
    await expect(cartPage.quantityInput).toHaveValue('2');
    await expect(cartPage.total('£110.00')).toBeVisible();

    await cartPage.setQuantity('0');
    await cartPage.update();
    await expect(cartPage.emptyMessage).toBeVisible();
    await expect(homePage.header.cartSummary(0)).toBeVisible();
  });
});


===== tests/e2e/storefront/navigation.spec.ts =====
import { test, expect } from '../../../src/fixtures/test.fixture';

const catalogProducts = [
  'Black heels',
  'Bronze sandals',
  'Brown Shades',
  'Grey jacket',
  'Noir jacket',
  'Striped top',
  'White sandals',
];

test.describe('Navigation', () => {
  test('navigates Catalog, product details, history, and Frontpage', async ({
    page, homePage, catalogPage, collectionsPage, productPage,
  }) => {
    await homePage.goto();
    await catalogPage.openFromSidebar();

    await expect(page).toHaveURL(/\/collections\/all/);
    await expect(catalogPage.heading).toBeVisible();

    if (await catalogPage.sortMenu.isVisible()) {
      await catalogPage.sortMenu.selectOption('price-ascending');
      await expect(page).toHaveURL(/sort_by=price-ascending/);

      const prices = (await catalogPage.productGrid.prices())
        .map((price) => Number(price.replace(/[^\d.]/g, '')));
      expect(prices).toEqual([...prices].sort((a, b) => a - b));
    }

    for (const name of catalogProducts) {
      await expect(catalogPage.productGrid.heading(name)).toBeVisible();
    }

    await catalogPage.productGrid.openProduct('Noir jacket');
    await expect(productPage.heading('Noir jacket')).toBeVisible();

    await page.goBack();
    await expect(page).toHaveURL(/\/collections\/all(?:\?.*)?$/);
    for (const name of catalogProducts) {
      await expect(catalogPage.productGrid.heading(name)).toBeVisible();
    }

    await collectionsPage.goto();
    await collectionsPage.openCollection('Frontpage');
    await expect(page).toHaveURL(/\/collections\/frontpage/);
    await expect(collectionsPage.productGrid.cards.first()).toBeVisible();
  });
});
